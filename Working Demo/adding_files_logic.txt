// --- Preload hierarchy and properties ---
let objectHierarchy = null;
let objectProperties = null;
let componentFragmentsMap = {}; // Maps component names to fragment IDs

// Load hierarchy and properties data
fetch('09_object_hierarchy.json')
    .then(response => response.json())
    .then(data => {
        objectHierarchy = data;
        logToConsole("Loaded object hierarchy");
    });

fetch('10_properties_all_objects.json')
    .then(response => response.json())
    .then(data => {
        objectProperties = data;
        logToConsole("Loaded object properties");
    });

// Create mapping between fragments and components
function buildComponentMap() {
    if (!objectProperties || !viewer.model) return;
    
    const tree = viewer.model.getInstanceTree();
    componentFragmentsMap = {};
    
    objectProperties.data.collection.forEach(item => {
        const dbId = item.objectid;
        const componentName = item.properties?.['Component Name'] || item.name;
        
        if (!componentName) return;
        
        tree.enumNodeFragments(dbId, fragId => {
            if (!componentFragmentsMap[componentName]) {
                componentFragmentsMap[componentName] = [];
            }
            componentFragmentsMap[componentName].push(fragId);
        });
    });
    
    logToConsole(`Mapped ${Object.keys(componentFragmentsMap).length} components`);
}

// Update on model load
function onDocumentLoadSuccess(doc) {
    // ... existing code ...
    viewer.addEventListener(Autodesk.Viewing.GEOMETRY_LOADED_EVENT, () => {
        logToConsole("Geometry loaded");
        populateFragmentCount();
        buildComponentMap(); // Build component mapping
    });
}

// Handle new command format (component-level animations)
function executeComponentAnimation(cmd) {
    if (cmd.component && componentFragmentsMap[cmd.component]) {
        const fragmentIds = componentFragmentsMap[cmd.component];
        return Promise.all(fragmentIds.map(fragId => 
            executeSmoothAnimation({...cmd, fragmentId: fragId})
        ));
    }
    return executeSmoothAnimation(cmd);
}

// Update animation execution

}
